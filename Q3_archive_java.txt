public class Q3 { 
    public static void main(String[] args) { 
        if (args.length == 0) { 
            System.out.println("Veuillez fournir un entier N en paramètre."); 
            return; 
        } 
    
        int n = Integer.parseInt(args[0]); 
        for (int i = n; i > 0; i--) { 
            System.out.print(i + (i > 0 ? " " : "")); 
        } 
        System.out.println(); 
    } 
}
